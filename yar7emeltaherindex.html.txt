        // =========================================================================
        // الحسابات المركزية الشاملة لجميع الشاشات
        // =========================================================================
        function updateAllCalculations() {
            // 1. قراءة وحساب المصاريف الأولية
            const vExp = parseFloat(document.getElementById('exp-vehicle').value) || 0;
            const fExp = parseFloat(document.getElementById('exp-fuel').value) || 0;
            const foodExp = parseFloat(document.getElementById('exp-food').value) || 0;
            const eqExp = parseFloat(document.getElementById('exp-equipment').value) || 0;
            const othExp = parseFloat(document.getElementById('exp-other').value) || 0;

            appState.expenses.vehicle = document.getElementById('exp-vehicle').value;
            appState.expenses.fuel = document.getElementById('exp-fuel').value;
            appState.expenses.food = document.getElementById('exp-food').value;
            appState.expenses.equipment = document.getElementById('exp-equipment').value;
            appState.expenses.other = document.getElementById('exp-other').value;
            appState.expenses.otherNote = document.getElementById('exp-other-note').value;

            const totalInitialExpenses = vExp + fExp + foodExp + eqExp + othExp;
            document.getElementById('total-initial-expenses').textContent = totalInitialExpenses.toFixed(2);
            document.getElementById('hint-initial-exp-in-harvest').textContent = totalInitialExpenses.toFixed(2);

            // 2. قراءة وحساب الجني والعمال + مرابيح البيع الفورية بالجلبة
            const ratePerJelba = parseFloat(document.getElementById('rate-per-jelba').value) || 0;
            const kgPerJelba = parseFloat(document.getElementById('kg-per-jelba').value) || 0;
            const sellPricePerJelba = parseFloat(document.getElementById('sell-price-per-jelba').value) || 0;

            appState.harvest.ratePerJelba = ratePerJelba;
            appState.harvest.kgPerJelba = kgPerJelba;
            appState.harvest.sellPricePerJelba = document.getElementById('sell-price-per-jelba').value;

            let totalJelba = 0;
            appState.harvest.groups.forEach((group, index) => {
                const j = parseFloat(group.jelba) || 0;
                totalJelba += j;
                const payEl = document.getElementById(`group-pay-${index}`);
                const kgEl = document.getElementById(`group-kg-${index}`);
                if (payEl) payEl.innerHTML = `${(j * ratePerJelba).toFixed(2)} <small class="text-[10px]">د.ت</small>`;
                if (kgEl) kgEl.innerHTML = `${(j * kgPerJelba).toFixed(0)} <small class="text-[10px]">كغ</small>`;
            });

            const totalHarvestKg = totalJelba * kgPerJelba;
            const totalWorkerWages = totalJelba * ratePerJelba;
            const totalJelbaSellRevenue = totalJelba * sellPricePerJelba;
            const instantJelbaProfit = totalJelbaSellRevenue - totalWorkerWages;
            const harvestNetAfterInitial = instantJelbaProfit - totalInitialExpenses;

            // تحديث مربع المرابيح الفوري في أعلى صفحة الجني
            document.getElementById('instant-jelba-gross').textContent = totalJelbaSellRevenue.toFixed(2);
            const instantProfitEl = document.getElementById('instant-jelba-profit');
            instantProfitEl.innerHTML = `${instantJelbaProfit.toFixed(2)} <small class="text-xs font-bold text-white">د.ت</small>`;

            const instantBox = document.getElementById('instant-sell-profit-box');
            if (instantJelbaProfit < 0 && sellPricePerJelba > 0) {
                instantBox.className = 'col-span-7 bg-gradient-to-br from-rose-700 to-rose-800 text-white rounded-xl p-2.5 shadow-sm flex flex-col justify-center';
            } else {
                instantBox.className = 'col-span-7 bg-gradient-to-br from-emerald-700 to-emerald-800 text-white rounded-xl p-2.5 shadow-sm flex flex-col justify-center';
            }

            // تحديث ملخص أسفل صفحة الجني
            document.getElementById('summary-total-jelba').textContent = totalJelba;
            document.getElementById('summary-total-kg').textContent = totalHarvestKg.toFixed(0);
            document.getElementById('summary-total-wages').textContent = totalWorkerWages.toFixed(2);
            document.getElementById('summary-jelba-revenue').textContent = totalJelbaSellRevenue.toFixed(2);
            document.getElementById('hint-harvest-kg').textContent = totalHarvestKg.toFixed(0);

            const harvestInstantSummaryEl = document.getElementById('harvest-instant-profit-summary');
            harvestInstantSummaryEl.textContent = `${instantJelbaProfit.toFixed(2)} د.ت`;
            harvestInstantSummaryEl.className = instantJelbaProfit >= 0 ? 'text-xl font-black text-emerald-400' : 'text-xl font-black text-rose-400';

            const harvestAfterInitialEl = document.getElementById('harvest-net-after-initial');
            harvestAfterInitialEl.textContent = `${harvestNetAfterInitial.toFixed(2)} د.ت`;
            harvestAfterInitialEl.className = harvestNetAfterInitial >= 0 ? 'font-extrabold text-gold-400' : 'font-extrabold text-rose-400';

            // 3. قراءة وحساب المعصرة والقطوع
            const oliveKg = parseFloat(document.getElementById('press-olive-kg').value) || 0;
            const oilLiters = parseFloat(document.getElementById('press-oil-liters').value) || 0;
            const pressingCost = parseFloat(document.getElementById('press-cost').value) || 0;

            appState.pressing.oliveKg = document.getElementById('press-olive-kg').value;
            appState.pressing.oilLiters = document.getElementById('press-oil-liters').value;
            appState.pressing.pressingCost = document.getElementById('press-cost').value;

            let yieldPercent = 0;
            let kgPerLiter = 0;
            if (oliveKg > 0 && oilLiters > 0) {
                yieldPercent = (oilLiters / oliveKg) * 100;
                kgPerLiter = oliveKg / oilLiters;
            }

            document.getElementById('yield-percentage').textContent = yieldPercent.toFixed(1);
            document.getElementById('kg-per-liter').textContent = kgPerLiter.toFixed(1);

            const ratingEl = document.getElementById('yield-rating');
            if (yieldPercent === 0) {
                ratingEl.textContent = 'بانتظار البيانات';
            } else if (yieldPercent >= 22) {
                ratingEl.textContent = 'ممتاز ما شاء الله 🌟';
            } else if (yieldPercent >= 18) {
                ratingEl.textContent = 'جيد جداً 👍';
            } else {
                ratingEl.textContent = 'متوسط 👌';
            }

            // 4. قراءة وحساب المرابيح الصافية للمشروع
            const sellPriceLiter = parseFloat(document.getElementById('sell-price-per-liter').value) || 0;
            const extraProjectCosts = parseFloat(document.getElementById('extra-project-costs').value) || 0;
            const otherIncome = parseFloat(document.getElementById('other-income').value) || 0;

            appState.profit.sellPricePerLiter = document.getElementById('sell-price-per-liter').value;
            appState.profit.extraProjectCosts = document.getElementById('extra-project-costs').value;
            appState.profit.otherIncome = document.getElementById('other-income').value;

            document.getElementById('profit-oil-liters-display').textContent = oilLiters;
            document.getElementById('profit-jelba-count-display').textContent = `${totalJelba} جلبة`;
            document.getElementById('profit-jelba-price-display').textContent = `${sellPricePerJelba.toFixed(2)} د.ت`;

            let primaryRevenue = 0;
            if (appState.profit.revenueMode === 'oil') {
                primaryRevenue = oilLiters * sellPriceLiter;
            } else {
                primaryRevenue = totalJelbaSellRevenue;
            }

            const totalRevenue = primaryRevenue + otherIncome;
            const totalPressingAndExtras = pressingCost + extraProjectCosts;
            const totalAllCosts = totalInitialExpenses + totalWorkerWages + totalPressingAndExtras;
            const netProfit = totalRevenue - totalAllCosts;

            document.getElementById('total-revenue').textContent = totalRevenue.toFixed(2);
            document.getElementById('breakdown-initial').textContent = totalInitialExpenses.toFixed(2);
            document.getElementById('breakdown-wages').textContent = totalWorkerWages.toFixed(2);
            document.getElementById('breakdown-pressing').textContent = totalPressingAndExtras.toFixed(2);
            document.getElementById('total-all-costs').textContent = totalAllCosts.toFixed(2);

            document.getElementById('final-net-profit').textContent = netProfit.toFixed(2);

            const finalCard = document.getElementById('final-profit-card');
            const statusText = document.getElementById('profit-status-text');

            if (netProfit > 0) {
                finalCard.className = 'bg-emerald-700 text-white rounded-2xl p-6 shadow-lg text-center space-y-1 transition-colors duration-300';
                statusText.textContent = 'مشروع رابح بإذن الله 🌿';
            } else if (netProfit < 0) {
                finalCard.className = 'bg-rose-700 text-white rounded-2xl p-6 shadow-lg text-center space-y-1 transition-colors duration-300';
                statusText.textContent = 'التكاليف تتجاوز المداخيل المسجلة حالياً';
            } else {
                finalCard.className = 'bg-olive-800 text-white rounded-2xl p-6 shadow-lg text-center space-y-1 transition-colors duration-300';
                statusText.textContent = 'نقطة التعادل (المداخيل تساوي المصاريف)';
            }

            saveState();
        }

        // جلب الوزن التقديري إلى صفحة المعصرة
        function syncHarvestWeight() {
            const totalKg = document.getElementById('summary-total-kg').textContent;
            document.getElementById('press-olive-kg').value = totalKg;
            updateAllCalculations();
            showToast(`تم نقل ${totalKg} كغ إلى حقل المعصرة`);
        }

        // مشاركة التلخيص عبر واتساب (WhatsApp Integration)
        // =========================================================================
        function shareToWhatsApp() {
            const today = new Date().toLocaleDateString('ar-TN');
            const rate = parseFloat(document.getElementById('rate-per-jelba').value) || 0;
            const sellJelbaPrice = parseFloat(document.getElementById('sell-price-per-jelba').value) || 0;
            const totalJelba = document.getElementById('summary-total-jelba').textContent;
            const totalKg = document.getElementById('summary-total-kg').textContent;
            const totalWages = document.getElementById('summary-total-wages').textContent;
            const jelbaRevenue = document.getElementById('summary-jelba-revenue').textContent;
            const instantProfit = document.getElementById('harvest-instant-profit-summary').textContent;

            let message = `🫒 *يرحم الطاهر - تطبيق جمع الزيتون* 🫒\n`;
            message += `📅 التاريخ: ${today}\n`;
            message += `━━━━━━━━━━━━━━━\n`;
            message += `👷‍♂️ *تفاصيل الجني والعمال* (أجرة الجلبة: ${rate} د.ت):\n`;

            let hasWorkers = false;
            appState.harvest.groups.forEach((g, i) => {
                const j = parseFloat(g.jelba) || 0;
                if (j > 0 || g.name.trim() !== '') {
                    hasWorkers = true;
                    const workerName = g.name.trim() ? g.name.trim() : `مجموعة ${i + 1}`;
                    const pay = (j * rate).toFixed(2);
                    message += `• ${workerName}: ${j} جلبة ⬅️ *${pay} د.ت*\n`;
                }
            });

            if (!hasWorkers) {
                message += `• لم يتم إدخال كميات بعد\n`;
            }

            message += `━━━━━━━━━━━━━━━\n`;
            message += `🧺 إجمالي الجلبات: *${totalJelba} جلبة*\n`;
            message += `⚖️ الوزن الكلي التقديري: *${totalKg} كغ*\n`;
            message += `💵 إجمالي أجور العمال: *${totalWages} د.ت*\n`;
            if (sellJelbaPrice > 0) {
                message += `🏷️ سعر الجلبة عند البيع: *${sellJelbaPrice} د.ت*\n`;
                message += `📈 إجمالي ثمن البيع: *${jelbaRevenue} د.ت*\n`;
                message += `💰 مرابيح البيع الصافية: *${instantProfit}*\n`;
            }
            message += `━━━━━━━━━━━━━━━\n`;
            message += `تم الإنشاء عبر تطبيق *يرحم الطاهر*`;

            const whatsappUrl = `https://wa.me/?text=${encodeURIComponent(message)}`;
            window.open(whatsappUrl, '_blank');
        }

        // =========================================================================
        // الحفظ والاسترجاع من LocalStorage
        // =========================================================================
        function saveState() {
            try {
                localStorage.setItem(STORAGE_KEY, JSON.stringify(appState));
            } catch (e) {
                console.error('Error saving state:', e);
            }
        }

        function loadState() {
            try {
                const saved = localStorage.getItem(STORAGE_KEY);
                if (saved) {
                    const parsed = JSON.parse(saved);
                    appState = { ...appState, ...parsed };
                }
            } catch (e) {
                console.error('Error loading state:', e);
            }
        }

        function populateInputsFromState() {

            document.getElementById('exp-fuel').value = appState.expenses.fuel || '';
            document.getElementById('exp-food').value = appState.expenses.food || '';

            document.getElementById('extra-project-costs').value = appState.profit.extraProjectCosts || '';
            document.getElementById('other-income').value = appState.profit.otherIncome || '';
        }

            localStorage.removeItem(STORAGE_KEY);
            appState = {
                expenses: { vehicle: '', fuel: '', food: '', equipment: '', other: '', otherNote: '' },
                harvest: {
                    ratePerJelba: 6,
                    kgPerJelba: 14,
                    sellPricePerJelba: '',
                    groups: [
                        { id: 1, name: '', jelba: '' },
                        { id: 2, name: '', jelba: '' },
                        { id: 3, name: '', jelba: '' },
                        { id: 4, name: '', jelba: '' }
                    ]
                },
                pressing: { oliveKg: '', oilLiters: '', pressingCost: '' },
                profit: { revenueMode: 'oil', sellPricePerLiter: '', extraProjectCosts: '', otherIncome: '' }

            renderGroups();
            updateAllCalculations();
            closeResetModal();
            showToast('تم تصفير جميع البيانات وبدء يوم جديد');
        }

        function showToast(msg) {
            const toast = document.getElementById('toast');
            toast.textContent = msg;
            toast.classList.remove('opacity-0');
            setTimeout(() => {
                toast.classList.add('opacity-0');
            }, 2500);
        }

        function escapeHtml(text) {
            if (!text) return '';
            return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
        }
